22/// Declare a linker set
23///
24/// # Examples
25///
26/// ```
27/// #![feature(const_mut_refs)]
28///
29/// linkset::declare!(TEST_SET: u32);
30///
31/// linkset::entry!(TEST_SET, u32, 1);
32/// linkset::entry!(TEST_SET, u32, 2);
33/// linkset::entry!(TEST_SET, u32, 3);
34///
35/// fn main() {
36///     for entry in &TEST_SET {
37///         println!("{entry}");
38///     }
39/// }
40/// ```
41#[allow(clippy::needless_doctest_main)]
42pub macro declare {
43    ($vis:vis $name:ident: $t:ty) => {
44        #[allow(non_upper_case_globals)]
45        $vis static $name: Linkset<$t> = declare!($name: $t |set| set);
46        declare_weak!($name);
47    },
48
49    ($name:ident: $t:ty |$set:ident| $value:expr) => {{
50       const _: () = {
51           #[export_name = concat!("__linkset_set_", stringify!($name))]
52           static GUARANTEE_SET_IS_UNIQUE: () = ();
53       };
54
55       #[allow(improper_ctypes)]
56       extern "C" {
57           #[link_name = concat!("__start___linkset_", stringify!($name))]
58           static mut set_start: Entry<$t>;
59           #[link_name = concat!("__stop___linkset_", stringify!($name))]
60           static mut set_stop: Entry<$t>;
61       }
62
63       let $set = unsafe {
64           Linkset::from_raw_parts(
65               core::ptr::addr_of_mut!(set_start),
66               core::ptr::addr_of_mut!(set_stop),
67           )
68       };
69
70       const fn check_linkset<T: LinksetOf<$t>>(set: T) -> T {
71           set
72       }
73       check_linkset($value)
74   }},
75}
76
77pub macro declare_weak($name:ident) {
78    ::core::arch::global_asm!(
79        concat!(".weak __start___linkset_", stringify!($name)),
80        concat!(".weak __stop___linkset_", stringify!($name)),
81    );
82}

84/// Declare an entry in a linker set
85///
86/// ```ignore
87/// entry!($set:ident: $t:ty, $value:expr);
88/// ```
89///
90/// The set must be of the same type `$t` and must be declared with [`linkset::declare!()`].
91/// `$set` must match the identifier used in `declare!()` and must be in scope (cannot be a path).
92pub macro entry($set:ident, $t:ty, $value:expr) {
93    const _: () = {
94        #[used]
95        #[link_section = concat!("__linkset_", stringify!($set))]
96        static ENTRY: Entry<$t> = {
97            const fn create_entry<T, S>(_: &S, value: &'static mut T) -> Entry<T>
98            where
99                S: LinksetOf<T>,
100            {
101                Entry::new(value)
102            }
103
104            create_entry(&$set, unsafe {
105                #[used]
106                static mut STORAGE: $t = $value;
107                &mut STORAGE
108            })
109        };
110    };
111}

113pub struct Linkset<T: 'static> {
114    start: *mut Entry<T>,
115    stop: *mut Entry<T>,
116}
117
118unsafe impl<T: Send + 'static> Send for Linkset<T> {}
119unsafe impl<T: Sync + 'static> Sync for Linkset<T> {}
120
121impl<T: 'static> Linkset<T> {
122    /// Create a `Linkset` from its raw parts
123    ///
124    /// This function not generally intended to be used directly. Instead, linker sets should
125    /// be create with the [`linkset::declare!()`] macro.
126    ///
127    /// # Safety
128    ///
129    #[inline]
130    pub const unsafe fn from_raw_parts(start: *mut Entry<T>, stop: *mut Entry<T>) -> Linkset<T> {
131        Self { start, stop }
132    }
133
134    #[inline]
135    pub const fn as_ptr(&self) -> *const Entry<T> {
136        self.start
137    }
138
139    #[inline]
140    pub const fn as_mut_ptr(&mut self) -> *mut Entry<T> {
141        self.start
142    }
143
144    #[inline]
145    pub const fn len(&self) -> usize {
146        unsafe { self.stop.offset_from(self.start) as usize }
147    }
148
149    #[inline]
150    pub const fn is_empty(&self) -> bool {
151        self.len() == 0
152    }
153
154    #[inline]
155    pub fn as_slice(&self) -> &'static [Entry<T>] {
156        unsafe { core::slice::from_raw_parts(self.as_ptr(), self.len()) }
157    }
158
159    #[inline]
160    pub fn as_mut_slice(&mut self) -> &'static mut [Entry<T>] {
161        unsafe { core::slice::from_raw_parts_mut(self.as_mut_ptr(), self.len()) }
162    }
163}
164
165impl<T: 'static> Deref for Linkset<T> {
166    type Target = [Entry<T>];
167
168    #[inline]
169    fn deref(&self) -> &Self::Target {
170        self.as_slice()
171    }
172}
173
174impl<T: 'static> DerefMut for Linkset<T> {
175    #[inline]
176    fn deref_mut(&mut self) -> &mut Self::Target {
177        self.as_mut_slice()
178    }
179}
180
181impl<T: 'static> IntoIterator for &Linkset<T> {
182    type IntoIter = core::slice::Iter<'static, Entry<T>>;
183    type Item = &'static Entry<T>;
184
185    #[inline]
186    fn into_iter(self) -> Self::IntoIter {
187        self.as_slice().iter()
188    }
